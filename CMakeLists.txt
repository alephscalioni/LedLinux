cmake_minimum_required(VERSION 3.20)
project(led-control VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
add_executable(led-control src/main.cpp)
target_link_libraries(led-control PRIVATE Qt6::Widgets Qt6::Network)
install(TARGETS led-control RUNTIME DESTINATION bin)
